## Webhook Processing (After Phase 3)

### Current (hardcoded Doftwerks)

```python
@frappe.whitelist(allow_guest=True, methods=["POST"])
def webhook():
    """Doftwerks webhook handler"""
    payload = json.loads(frappe.request.data or b"{}")
    event_type = payload.get("eventType")
    if event_type != "TransmissionStatusEvent":
        return  # Ignore
    
    data = payload.get("data", payload)
    irn = data.get("irn")
    status_code = data.get("receipt_status")
    
    # Doftwerks-specific status mapping
    status_labels = {1: "INITIATED", 2: "SIGNED", ...}
    status = status_labels.get(status_code)
    
    # Find invoice and update
    inv_name = db.get_value("Sales Invoice", {"nrs_irn": irn})
    db.set_value("Sales Invoice", inv_name, "nrs_receipt_status", status)
```

### After (provider-agnostic)

```python
@frappe.whitelist(allow_guest=True, methods=["POST"])
def webhook():
    """Multi-provider webhook handler"""
    settings = get_single("NRS E-Invoice Settings")
    provider = instantiate_provider(settings.provider)
    
    payload = json.loads(frappe.request.data or b"{}")
    signature = frappe.request.headers.get("X-Signature")
    
    # Let provider parse webhook
    result = provider.handle_webhook(payload, signature)
    
    if not result['valid']:
        return {"error": result['error']}
    
    # Standard result from any provider
    irn = result['irn']
    status = result['status']
    invoice_name = result['invoice_number']
    
    # Update invoice (same code for all providers)
    db.set_value("Sales Invoice", invoice_name, {
        "nrs_receipt_status": status,
        "nrs_error": ""
    })
```

**Benefit**: Webhook handling is provider-specific but integration is generic

## Status Reconciliation (Daily Job)

### Before (hardcoded Doftwerks lookup)

```python
def reconcile_transmissions():
    """Daily job: sync status from platform"""
    url = f"{base_url}/api/v1/einvoice/lookup/{irn}"
    response = requests.get(url, headers=doftwerks_headers)
    data = response.json()
    
    # Doftwerks returns transmit_status as lowercase
    status_map = {
        "initiated": "INITIATED",
        "signed": "SIGNED",
        "transmitted": "TRANSMITTED"
    }
    status = status_map[data['transmit_status'].lower()]
```

### After (provider-agnostic)

```python
def reconcile_transmissions():
    """Daily job: sync status from platform"""
    settings = get_single("NRS E-Invoice Settings")
    provider = instantiate_provider(settings.provider)
    
    result = provider.query_status(irn, credentials)
    
    if result['success']:
        # Same code for all providers
        db.set_value("Sales Invoice", name, "nrs_receipt_status", result['status'])
```

**Benefit**: Provider handles status format; reconciliation is generic

## Method Signatures (Reference)

### Core Provider Methods

```python
# Credentials
(bool, str) = provider.validate_credentials(
    credentials: Dict[str, Any]
)

# Transmission
result: Dict = provider.transmit(
    payload: Dict[str, Any],
    credentials: Dict[str, Any]
)
# Returns: {success, irn, status, qr_code, error, raw_response}

# Response Parsing
result: Dict = provider.parse_response(
    response: requests.Response
)
# Same return structure as transmit()

# UI Fields
fields: List[Dict] = provider.get_credential_fields()
# Returns: [{fieldname, label, fieldtype, required, ...}, ...]

# Webhook Processing
webhook_result: Dict = provider.handle_webhook(
    payload: Dict[str, Any],
    signature: Optional[str]
)
# Returns: {valid, invoice_number, irn, status, error}

# Status Query
status_result: Dict = provider.query_status(
    irn: str,
    credentials: Dict[str, Any]
)
# Returns: {success, status, error}
```
